独立复验 OSS（预期 3,762 / 3,608 / 95，散列与上面相同）：

```bash
mkdir -p /tmp/goodsdex-acceptance
for name in goods.json meta.json merge-audit.json; do
  curl --noproxy '*' -fsS "https://spend-good.oss-cn-shanghai.aliyuncs.com/data/public/$name" \
    -o "/tmp/goodsdex-acceptance/$name"
done
python3 - <<'PYCODE'
import hashlib, json
from pathlib import Path
p = Path('/tmp/goodsdex-acceptance')
raw = (p/'goods.json').read_bytes()
rows = json.loads(raw)
audit = json.loads((p/'merge-audit.json').read_text())
print(len(rows), len({r['id'] for r in rows}), len(audit['conflicts']))
assert len(rows) == 3762 and len({r['id'] for r in rows}) == 3608
assert len(audit['conflicts']) == 95
assert json.loads((p/'meta.json').read_text())['count'] == 3762
assert hashlib.sha256(raw).hexdigest() == audit['deduplication']['output_sha256']
print(hashlib.sha256(raw).hexdigest())
PYCODE
```
